from flask import Flask, jsonify, request
from flask_sqlalchemy import SQLAlchemy

app = Flask(__name__)

# ตั้งค่าฐานข้อมูล SQLite (สร้างไฟล์ restaurants.db อัตโนมัติ)
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///restaurants.db'
app.config['SQLALCHEMY_TRACK_MODIFICATION'] = False
db = SQLAlchemy(app)

# ==========================================
# Database Model (โครงสร้างตารางเก็บข้อมูล)
# ==========================================
class Restaurant(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    name = db.Column(db.String(100), nullable=False)
    open_time = db.Column(db.String(10), nullable=False)   # เช่น "10:00"
    close_time = db.Column(db.String(10), nullable=False)  # เช่น "21:00"
    avg_price = db.Column(db.Float, nullable=False)
    category = db.Column(db.String(50), nullable=False)
    location = db.Column(db.String(200), nullable=False)
    is_open = db.Column(db.Boolean, default=True)

    def to_dict(self):
        """แปลงข้อมูลเป็น Dictionary เพื่อส่งออกเป็น JSON"""
        return {
            "id": self.id,
            "name": self.name,
            "open_time": self.open_time,
            "close_time": self.close_time,
            "avg_price": self.avg_price,
            "category": self.category,
            "location": self.location,
            "is_open": self.is_open
        }

# สร้างตารางใน Database เมื่อเริ่มต้นรันแอป
with app.app_context():
    db.create_all()

# ==========================================
# API Routes (6 ฟังก์ชันตามโจทย์)
# ==========================================

# 1. เพิ่มร้านอาหาร (POST /api/restaurants)
@app.route('/api/restaurants', methods=['POST'])
def add_restaurant():
    data = request.get_json()
    
    # เช็กว่าส่งข้อมูลมาครบหรือไม่
    required_fields = ['name', 'open_time', 'close_time', 'avg_price', 'category', 'location']
    if not data or not all(k in data for k in required_fields):
        return jsonify({"error": "กรุณากรอกข้อมูลให้ครบถ้วน"}), 400

    new_restaurant = Restaurant(
        name=data['name'],
        open_time=data['open_time'],
        close_time=data['close_time'],
        avg_price=data['avg_price'],
        category=data['category'],
        location=data['location'],
        is_open=data.get('is_open', True)
    )
    
    db.session.add(new_restaurant)
    db.session.commit()
    
    return jsonify({
        "message": "เพิ่มร้านอาหารเรียบร้อยแล้ว",
        "data": new_restaurant.to_dict()
    }), 201


# 2. แก้ไขข้อมูลร้าน (PUT /api/restaurants/<id>)
@app.route('/api/restaurants/<int:restaurant_id>', methods=['PUT'])
def edit_restaurant(restaurant_id):
    restaurant = Restaurant.query.get(restaurant_id)
    if not restaurant:
        return jsonify({"error": "ไม่พบร้านค้านี้"}), 404
        
    data = request.get_json()
    if not data:
        return jsonify({"error": "ไม่มีข้อมูลที่ต้องการแก้ไข"}), 400

    # อัปเดตข้อมูลเฉพาะฟิลด์ที่ส่งมา
    restaurant.name = data.get('name', restaurant.name)
    restaurant.open_time = data.get('open_time', restaurant.open_time)
    restaurant.close_time = data.get('close_time', restaurant.close_time)
    restaurant.avg_price = data.get('avg_price', restaurant.avg_price)
    restaurant.category = data.get('category', restaurant.category)
    restaurant.location = data.get('location', restaurant.location)
    
    db.session.commit()
    
    return jsonify({
        "message": f"แก้ไขข้อมูลร้าน ID {restaurant_id} เรียบร้อยแล้ว",
        "data": restaurant.to_dict()
    }), 200


# 3. ลบร้าน (DELETE /api/restaurants/<id>)
@app.route('/api/restaurants/<int:restaurant_id>', methods=['DELETE'])
def delete_restaurant(restaurant_id):
    restaurant = Restaurant.query.get(restaurant_id)
    if not restaurant:
        return jsonify({"error": "ไม่พบร้านค้านี้"}), 404

    db.session.delete(restaurant)
    db.session.commit()
    
    return jsonify({"message": f"ลบร้าน '{restaurant.name}' (ID: {restaurant_id}) เรียบร้อยแล้ว"}), 200


# 4. ค้นหาร้าน (GET /api/restaurants/search?q=...)
@app.route('/api/restaurants/search', methods=['GET'])
def search_restaurants():
    keyword = request.args.get('q', '')
    if not keyword:
        return jsonify({"error": "กรุณาระบุคำค้นหา (เช่น ?q=ญี่ปุ่น)"}), 400

    # ค้นหาจาก ชื่อ, ประเภทอาหาร หรือ ตำแหน่ง
    search_filter = f"%{keyword}%"
    results = Restaurant.query.filter(
        (Restaurant.name.like(search_filter)) |
        (Restaurant.category.like(search_filter)) |
        (Restaurant.location.like(search_filter))
    ).all()

    return jsonify({
        "query": keyword,
        "count": len(results),
        "results": [res.to_dict() for res in results]
    }), 200


# 5. ดูรายละเอียดร้าน (GET /api/restaurants/<id>)
@app.route('/api/restaurants/<int:restaurant_id>', methods=['GET'])
def get_restaurant_details(restaurant_id):
    restaurant = Restaurant.query.get(restaurant_id)
    if not restaurant:
        return jsonify({"error": "ไม่พบร้านค้านี้"}), 404

    return jsonify(restaurant.to_dict()), 200


# 6. เปิด/ปิดสถานะร้าน (PATCH /api/restaurants/<id>/toggle)
@app.route('/api/restaurants/<int:restaurant_id>/toggle', methods=['PATCH'])
def toggle_restaurant_status(restaurant_id):
    restaurant = Restaurant.query.get(restaurant_id)
    if not restaurant:
        return jsonify({"error": "ไม่พบร้านค้านี้"}), 404

    data = request.get_json() or {}
    
    # ถ้าส่งค่า is_open มา ให้ใช้ค่านั้น / ถ้าไม่ส่งมา ให้สลับสถานะเดิมอัตโนมัติ
    if 'is_open' in data:
        restaurant.is_open = bool(data['is_open'])
    else:
        restaurant.is_open = not restaurant.is_open

    db.session.commit()
    
    status_text = "เปิดร้าน" if restaurant.is_open else "ปิดร้าน"
    return jsonify({
        "message": f"เปลี่ยนสถานะร้าน '{restaurant.name}' เป็น: {status_text}",
        "data": restaurant.to_dict()
    }), 200


if __name__ == '__main__':
    app.run(debug=True, port=5000)
